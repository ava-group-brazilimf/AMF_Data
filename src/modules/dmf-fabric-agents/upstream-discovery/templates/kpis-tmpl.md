  header:
    title: "Key Performance Indicators (KPIs)"

      - field: "Project"
        required: true
      - field: "Date"
        default: "current"
      - field: "Author"
        default: "DataStrategist"
      - field: "Version"
        default: "1.0"

  summary:
    title: "KPI Summary"
    format: "table"
    columns:
      - name: "#"
        width: 5
      - name: "KPI Name"
        width: 30
      - name: "Baseline"
        width: 15
      - name: "Target"
        width: 15
      - name: "Timeline"
        width: 15

  kpi_detail:
    title: "KPI Details"
    repeat_for_each_kpi: true
    subsections:
      - name: "header"
        format: "### KPI {n}: {kpi_name}"
      
      - name: "category"
        label: "Category"
        options:
          - "Financial"
          - "Operational"
          - "Customer"
          - "Compliance"
          - "Data Quality"
      
      - name: "attributes"
        format: "table"
        fields:
          - name: "Description"
            required: true
            prompt: "What does this KPI measure?"
          - name: "Baseline"
            required: true
            prompt: "What is the current value?"
          - name: "Target"
            required: true
            prompt: "What is the goal value?"
          - name: "Unit"
            required: true
            options: ["percentage", "count", "currency", "time", "ratio"]
          - name: "Direction"
            required: true
            options: ["Higher is better", "Lower is better"]
      
      - name: "calculation"
        title: "Calculation"
        format: "code_block"
        required: true
        prompt: "What is the formula or calculation method?"
      
      - name: "data_sources"
        title: "Data Sources"
        format: "bullet_list"
        required: true
        min_items: 1
        format_each: "{source_name} - {field/table}"
      
      - name: "measurement"
        title: "Measurement"
        fields:
          - name: "Frequency"
            required: true
            options: ["Real-time", "Daily", "Weekly", "Monthly", "Quarterly"]
          - name: "Owner"
            required: false
            prompt: "Who is responsible for this KPI?"

  measurement_plan:
    title: "Measurement Plan"
    subsections:
      - name: "data_collection"
        title: "Data Collection"
        format: "table"
        columns: ["KPI", "Source System", "Collection Method", "Frequency"]
      
      - name: "reporting"
        title: "Reporting Cadence"
        format: "table"
        columns: ["Report", "KPIs Included", "Frequency", "Audience"]

  baseline_actions:
    title: "Baseline Measurement Actions"
    format: "table"
    columns: ["KPI", "Current Status", "Action Needed", "Due Date"]
    show_if: "any baseline is TBD"

  risks_assumptions:
    title: "Risks and Assumptions"
    subsections:
      - name: "risks"
        title: "Risks"
        format: "bullet_list"
      - name: "assumptions"
        title: "Assumptions"
        format: "bullet_list"

  next_steps:
    title: "Next Steps"
    default_content: |
      1. Validate KPIs with stakeholders
      2. Define success criteria - `*success-criteria`
      3. Pass to BusinessAnalyst for STTM creation

  footer:
    content: "*Document generated by DataStrategist Agent*"
    timestamp: true

kpi_schema:
  required_fields:
    - name
    - description
    - baseline
    - target
    - calculation
    - data_source
  
  optional_fields:
    - frequency
    - owner
    - timeline
    - unit
    - direction

validation_rules:
  - name: "minimum_kpis"
    check: "kpi_count >= 3"
    severity: "error"
    message: "At least 3 KPIs required"
  
  - name: "baseline_present"
    check: "each kpi has baseline"
    severity: "error"
    message: "All KPIs must have a baseline value"
  
  - name: "target_present"
    check: "each kpi has target"
    severity: "error"
    message: "All KPIs must have a target value"
  
  - name: "calculation_documented"
    check: "each kpi has calculation"
    severity: "error"
    message: "All KPIs must have calculation method"
  
  - name: "data_source_identified"
    check: "each kpi has data_source"
    severity: "error"
    message: "All KPIs must identify data sources"
